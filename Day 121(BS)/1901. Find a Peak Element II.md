class Solution {
    public int[] findPeakGrid(int[][] mat) {
        int low=0;
        int high=mat[0].length-1;
        while(low<=high){
            int mid= low+(high-low)/2;
            int row=findMaxRow(mat,mid);
            if(mid-1>=0 && mat[row][mid]<mat[row][mid-1]){
                high=mid-1;
            }
            else if(mid+1<mat[0].length && (mat[row][mid]<mat[row][mid+1])){
                low=mid+1;
            }
            else{
                return new int[]{row,mid};
            }
        }
        return  new int[]{-1,-1};
    }
    public int findMaxRow(int[][] mat,int col){
        int max=0;
        int row=0;
        for(int i=0;i<mat.length;i++){
            if(max<mat[i][col]){
                max=mat[i][col];
                row=i;
            }
        }
        return row;
    }
}
