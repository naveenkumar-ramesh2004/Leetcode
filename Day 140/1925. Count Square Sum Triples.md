## [1925. Count Square Sum Triples](https://leetcode.com/problems/count-square-sum-triples/description/)
```
class Solution {
    public int countTriples(int n) {
        int count=0;
        for(int a=1;a<n;a++){
            for(int b=a+1;b<n;b++){
                int c =(int) Math.sqrt(a*a+b*b);
                if(c<=n && a*a+b*b==c*c) count+=2;
            }
        }
        return count;
    }
}
