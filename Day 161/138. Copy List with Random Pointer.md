class Solution {
    public Node copyRandomList(Node head) {
        Node temp=head;
        while(temp!=null){
            Node newNode = new Node(temp.val);
            newNode.next=temp.next;
            temp.next=newNode;
            temp=newNode.next;
        }
        temp=head;
        while(temp!=null){
            temp.next.random=(temp.random==null)?null:temp.random.next;
            temp=temp.next.next;
        }
        temp=head;
        Node result=new Node(-1);
        Node res=result;
        while(temp!=null){
            res.next=temp.next;
            res=temp.next;
            temp.next=temp.next.next;
            temp=temp.next;
        }
        return result.next;
    }
}
