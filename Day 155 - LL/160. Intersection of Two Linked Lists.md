// start compare from same length
public class Solution {
    public ListNode getIntersectionNode(ListNode headA, ListNode headB) {
        int length1=length(headA);
        int length2=length(headB);
        int move=Math.abs(length1-length2);
        if(length1>length2){
            headA=movehead(headA,move);
        }
        else{
            headB=movehead(headB,move);
        }
        while(headA!=null && headB!=null){
            if(headA==headB) return headA;
            headA=headA.next;
            headB=headB.next;
        }
        return null;
        
    }
    public int length(ListNode head){
        int len=0;
        while(head!=null){
            len++;
            head=head.next;
        }
        return len;
    }
    public ListNode movehead(ListNode head,int move){
        for(int i=0;i<move;i++){
            head=head.next;
        }
        return head;
    }
}
