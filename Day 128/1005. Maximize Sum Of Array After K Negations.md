class Solution {
    public int largestSumAfterKNegations(int[] nums, int k) {
        Arrays.sort(nums);
        int i=0,smallidx=0;
        while (i < nums.length && nums[i] < 0 && k > 0) {
            nums[i] = -nums[i];
            i++;
            k--; 
        }

        if(k%2==1){
            for(i=0;i<nums.length;i++){
                if(nums[i]<=nums[smallidx]) smallidx=i;
            }
            nums[smallidx]=-nums[smallidx];
        }
        int sum=0;
        for(int val:nums) sum+=val;
        return sum;
    }
}
