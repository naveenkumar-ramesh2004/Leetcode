## [204. Count Primes](https://leetcode.com/problems/count-primes/description/)
```
class Solution {
    public int countPrimes(int n) {
        boolean[] notPrime= new boolean[n];
        for(int i=2;i*i<n;i++){
            if(!notPrime[i]){
                int j=i*i;
                while(j<n){
                    notPrime[j]=true;
                    j+=i;
                }
            }
        }
        int count=0;
        for(int i=2;i<n;i++){
            if(!notPrime[i]) count++;
        }
        return count;
    }
}
