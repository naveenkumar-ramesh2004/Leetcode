//Insted use StartsWith()
class Solution {
    public String longestCommonPrefix(String[] strs) {
        String temp = strs[0];
        while (!temp.isEmpty()) {
            boolean flag = true;
            for (int i = 0; i < strs.length; i++) {
                if(!strs[i].startsWith(temp)){
                    flag=false;
                    break;
                }
            }
            if (flag)
                return temp;
            else {
                temp = temp.substring(0, temp.length() - 1);
            }
        }
        return "";
    }
}
