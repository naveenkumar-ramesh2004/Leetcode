// optimal use Fixed two element + use two pointer
class Solution {
    public List<List<Integer>> fourSum(int[] nums, int target) {
        Arrays.sort(nums);
        List<List<Integer>> ans = new ArrayList<>();
        for(int i=0;i<nums.length;i++){
            if(i>0 && nums[i]==nums[i-1]) continue;
            for(int j=i+1;j<nums.length;j++){
                if(j>i+1 && nums[j]==nums[j-1]) continue; 
                int k=j+1;
                int q=nums.length-1;
                while(k<q){
                    long sum=(long)nums[i]+nums[j]+nums[k]+nums[q];
                    if(sum<target){
                        k++;
                    }
                    else if(sum>target){
                        q--;
                    }
                    else{
                        List<Integer> temp = Arrays.asList(nums[i],nums[j],nums[k],nums[q]);
                        ans.add(temp);
                        k++;
                        q--;
                        while(k<q && nums[k]==nums[k-1]) k++;
                        while(k<q && nums[q]==nums[q+1]) q--;
                    }
                }

            }
        }
        return ans;
    }
}
