// Make-and-Break Circle technique
class Solution {
    public ListNode rotateRight(ListNode head, int k) {
        if(head==null || head.next==null || k==0) return head;
        ListNode tail =head;
        int len=1;
        while(tail.next!=null){
            tail=tail.next;
            len++;
        }
        k=k%len;
        if(k==0) return head;
        tail.next=head;
        ListNode newtail=head;
        for(int i=1;i<len-k;i++){
            newtail=newtail.next;
        }
        head=newtail.next;
        newtail.next=null;
        return head;
    }
}
