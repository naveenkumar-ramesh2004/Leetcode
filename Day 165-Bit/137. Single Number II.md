// We track how many times each bit has appeared:

// ones → bits that appeared 1 time
// twos → bits that appeared 2 times
// When a bit appears 3 times, remove it from both.


class Solution {
    public int singleNumber(int[] nums) {
        int one=0;
        int two=0;
        for(int n:nums){
            one=(one^n)&~two;
            two=(two^n)&~one;
        }
        return one;
    }
}
